```python
# apps/image_search/services/storage_utils.py

def build_seo_filename(product: "Product", index: int, ext: str = "jpg") -> str:
    brand   = slugify_ru(product.brand)
    article = slugify_ru(product.article or "no-article")
    desc    = slugify_ru(product.name, max_words=5)
    uid     = hashlib.md5(f"{product.pk}-{index}".encode()).hexdigest()[:8]
    return f"{brand}-{article}-{desc}-{uid}.{ext}"

def build_s3_path(product: "Product", filename: str) -> str:
    brand   = slugify_ru(product.brand)
    article = slugify_ru(product.article or "no-article")
    return f"products/{brand}/{article}/{filename}"
```

### 2.4 Три версии файла

| Суффикс | Разрешение | Качество | Хранится в |
|---|---|---|---|
| (оригинал) | max 1600px | 90% | `ProductImage.s3_key` |
| `_preview` | max 600px | 85% | `ProductImage.s3_key_preview` |
| `_thumb` | max 150px | 80% | `ProductImage.s3_key_thumb` |

Resize через `Pillow` — переиспользуем `_resize()` и `_to_jpeg_bytes()` из `apps/products/storage.py`.

### 2.5 Очистка при удалении

```python
# apps/image_search/signals.py

@receiver(post_delete, sender=ProductImage)
def delete_image_from_s3(sender, instance, **kwargs):
    """Удаляет все 3 версии файла из S3."""
    storage = default_storage
    for key_field in ['s3_key', 's3_key_preview', 's3_key_thumb']:
        key = getattr(instance, key_field, '')
        if key:
            try:
                storage.delete(key)
            except Exception:
                logger.warning(f"S3 delete failed: {key}")
```

**Важно:** Сигнал на `ProductImage`, **не** на `Listing`. Удаление Listing не должно удалять фото товара.

Django Admin bulk delete:
```python
class ProductImageAdmin(admin.ModelAdmin):
    def delete_queryset(self, request, queryset):
        for obj in queryset:
            obj.delete()  # вызовет post_delete сигнал
```

```python
    source_id:   str = ""

    requires_key: bool = False

    def __init__(self, product: "Product"):
        self.product = product

    @abstractmethod
    def search(self) -> list[ImageCandidate]: ...

    def is_available(self) -> bool:
        return True

    def build_queries(self) -> list[str]:
        return _default_queries(self.product)
```

### 3.2 Реестр

```python
# apps/image_search/sources/registry.py

def get_active_sources(product: "Product") -> list[BaseImageSource]:
    enabled = settings.IMAGE_SOURCES_ENABLED
    sources = []
    for source_id in enabled:
        cls = _REGISTRY.get(source_id)
        if cls:
            instance = cls(product)
            if instance.is_available():
                sources.append(instance)
    return sorted(sources, key=lambda s: s.tier)
```

## 4. Стратегия запросов

### 4.1 QueryBuilder (реальные имена полей)

```python
# apps/image_search/services/query_builder.py

def build_queries(product: "Product") -> list[tuple[str, str]]:
    queries = []
    pn  = product.article.strip() if product.article else ""
    mfr = product.brand.strip()
    nom = product.name.strip()

    if pn and not _is_unreliable_article(pn, mfr):
        queries.append((f"{pn} {mfr}", "HIGH"))
        queries.append((f"{pn} автозапчасть", "HIGH"))

    clean_pn = re.sub(r'[^A-Za-z0-9]', '', pn)
    if clean_pn and clean_pn != pn:
        queries.append((f"{clean_pn} {mfr}", "MEDIUM"))

    desc_words = _extract_keywords(nom, max_words=5)
    if desc_words:
        queries.append((f"{mfr} {desc_words}", "LOW"))
        queries.append((desc_words, "LOW"))

    return queries or [(nom, "LOW")]
```

### 4.2 Уровень уверенности

| Условие | `search_confidence` | Влияние на score |
|---|---|---|
| HIGH + Tier 1–2 | `HIGH` | Полный score |
| HIGH + Tier 3–4 | `MEDIUM` | −0.05 |
| MEDIUM запрос | `MEDIUM` | −0.10 |
| LOW (описание) | `LOW` | −0.20, статус `NEEDS_REVIEW` |
| Нет артикула | `VERY_LOW` | −0.25, статус `LOW_CONFIDENCE` |

## 5. Quality Scoring

### 5.1 Критерии

| Критерий | Вес | Логика |
|---|---|---|
| Разрешение | 0.30 | <300px→0 / 300–600→0.5 / >800→1.0 |
| Источник | 0.25 | Tier1→1.0 / Tier2→0.8 / Tier3→0.6 / Tier4→0.4 |
| Уверенность | 0.20 | HIGH→1.0 / MEDIUM→0.7 / LOW→0.4 / VERY_LOW→0.2 |
| Пропорции | 0.15 | 1:1→1.0 / 4:3→0.9 / прочее→0.6 |
| Размер файла | 0.10 | <10KB→0.1 / 50KB–2MB→1.0 / >5MB→0.3 |

### 5.2 Пороги

```
≥ 0.75 → AUTO_APPROVED (зелёный)
0.50–0.74 → NEEDS_REVIEW (жёлтый)
< 0.50 → LOW_CONFIDENCE (красный)
не найдено → NO_IMAGE (серый) → product.image_status = "no_image"
```

```python
class ImageSearchPipeline:
    def __init__(self, product: "Product"):
        self.product = product
        self.tenant = product.tenant      # ← tenant через product
        ...

    def _download_and_save(self, candidate, position):
        # Используем default_storage (YC S3, уже настроен)
        from django.core.files.storage import default_storage

        seo_name = build_seo_filename(self.product, position)
        s3_path  = build_s3_path(self.product, seo_name)

        # 3 версии через Pillow
        original_bytes = _to_jpeg_bytes(_resize(img.copy(), 1600))

        default_storage.save(s3_key_preview, BytesIO(preview_bytes))

        record = ProductImage(
            product=self.product,

            s3_key_thumb=s3_key_thumb,

            source_id=candidate.source_id,

            quality_score=candidate.quality_score,
```

## 8. Celery-задачи

```python
# apps/image_search/tasks.py

@shared_task(bind=True, max_retries=3, default_retry_delay=120,
             queue="image_search")
def search_images_for_product(self, product_id: int) -> dict:
    product = Product.objects.select_related('tenant').get(pk=product_id)
    good_count = product.images.filter(
        status__in=["auto_approved", "manually_set", "imported"]
    ).count()
    if good_count >= settings.IMAGE_SEARCH_SETTINGS["MAX_IMAGES_PER_PRODUCT"]:
        return {"skipped": True}
    return ImageSearchPipeline(product).run()

@shared_task(queue="image_search_bulk")
def search_images_bulk(product_ids: list[int], delay_sec: float = 2.0):
    chunk_size = 50
    for chunk_start in range(0, len(product_ids), chunk_size):
        chunk = product_ids[chunk_start:chunk_start + chunk_size]
        for i, pid in enumerate(chunk):
            countdown = (chunk_start + i) * delay_sec
            search_images_for_product.apply_async(
                args=[pid], countdown=countdown, queue="image_search")

@shared_task(queue="image_search")
def refresh_stale_images():
    """Beat: раз в ночь, max 500 товаров."""
    pids = list(
        Product.objects.filter(
            models.Q(images__isnull=True) |
            models.Q(images__status="low_confidence")
        ).distinct().values_list("id", flat=True)[:500]
    )
    if pids:
        search_images_bulk.delay(pids, delay_sec=3.0)
```

## Что изменилось vs ТЗ v2

| Аспект | v2 | v3 (исправлено) |
|---|---|---|
| Хранилище | Cloudflare R2 | **Yandex Cloud S3** |
| ProductImage | Новая модель в `app/images/` | **Расширение существующей** в `apps/products/` |
| Приложение | `app/images/` | **`apps/image_search/`** |
| Имена полей | manufacturer, part_number, nomenclature | **brand, article, name** |
| Trust Level | §11 интеграция | **Убран из MVP** |

| Celery task paths | `app.tasks.*` | **`apps.image_search.tasks.*`** |
| Celery worker | Не указан | **Отдельный `celery_worker_images`** |
| Сигнал удаления | На Listing + Product | **Только на ProductImage** |
| HTTP-клиент | httpx (новый) | **httpx** (добавляем) |
| Status default | needs_review | **imported** (обратная совместимость) |
